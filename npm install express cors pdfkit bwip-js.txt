import express from 'express';
import cors from 'cors';
import PDFDocument from 'pdfkit';
import bwipjs from 'bwip-js';

const app = express();
app.use(cors());

const LIGHTSPEED_DOMAIN = process.env.LIGHTSPEED_DOMAIN;
const LIGHTSPEED_TOKEN = process.env.LIGHTSPEED_TOKEN;

// The layout template you designed, mapped in PDF points
const labelTemplate = {
  elements: [
    { type: "text", field: "name", x: 5, y: 5, fontSize: 8, maxWidth: 179, align: "left" },
    { type: "barcode", field: "sku", x: 25, y: 20, width: 140, height: 28 },
    { type: "text", field: "price", x: 130, y: 55, fontSize: 12, bold: true },
    { type: "text", field: "sku", x: 25, y: 55, fontSize: 7, bold: false }
  ]
};

app.get('/api/labels/po/:poId/pdf', async (req, res) => {
  try {
    const { poId } = req.params;
    
    // 1. Fetch the Purchase Order data
    const poResponse = await fetch(`https://${LIGHTSPEED_DOMAIN}.retail.lightspeed.app/api/2.0/consignments/${poId}/products`, {
      headers: { 'Authorization': `Bearer ${LIGHTSPEED_TOKEN}` }
    });
    const poData = await poResponse.json();

    // 2. Initialize the PDF Document (US Letter Size)
    const doc = new PDFDocument({
      size: 'letter',
      margins: { top: 36, bottom: 36, left: 13.5, right: 13.5 }, // 0.5" top/bottom, 0.19" sides
      autoFirstPage: true
    });

    // 3. Force the browser to instantly download the PDF
    res.setHeader('Content-Type', 'application/pdf');
    res.setHeader('Content-Disposition', `attachment; filename="PO_${poId}_Labels.pdf"`);
    doc.pipe(res);

    let labelCount = 0;

    // 4. Process each line item
    for (const item of poData.data) {
      const qty = item.received || item.count || 1;
      
      // Fetch full product details to get name, SKU, price
      const prodResponse = await fetch(`https://${LIGHTSPEED_DOMAIN}.retail.lightspeed.app/api/2.0/products/${item.product_id}`, {
        headers: { 'Authorization': `Bearer ${LIGHTSPEED_TOKEN}` }
      });
      const prodData = await prodResponse.json();
      const product = prodData.data;
      
      const valuesMap = {
        name: product.name || '',
        sku: product.sku || 'UNKNOWN',
        price: `$${product.price_including_tax || '0.00'}`
      };

      // Generate a barcode buffer once per product to save processing time
      let barcodeBuffer = null;
      if (valuesMap.sku !== 'UNKNOWN') {
        barcodeBuffer = await bwipjs.toBuffer({
          bcid: 'code128',       // Standard retail barcode format
          text: valuesMap.sku, 
          scale: 3, 
          height: 10, 
          includetext: false     // We draw the text manually via PDFKit
        });
      }

      // 5. Draw the necessary quantity of labels for this item
      for (let i = 0; i < qty; i++) {
        // Trigger a new page if the current one hits 30 labels
        if (labelCount > 0 && labelCount % 30 === 0) {
          doc.addPage();
        }

        // --- THE GRID MATH ---
        const positionOnPage = labelCount % 30;
        const col = positionOnPage % 3;           // 0, 1, or 2
        const row = Math.floor(positionOnPage / 3); // 0 through 9

        // 13.5 (Left Margin) + Column * (189 Label Width + 9 Gap)
        const originX = 13.5 + (col * 198); 
        // 36 (Top Margin) + Row * (72 Label Height)
        const originY = 36 + (row * 72); 
        
        // 6. Draw elements from the template relative to the origin
        for (const el of labelTemplate.elements) {
          const val = valuesMap[el.field];
          
          if (el.type === 'text') {
            doc.fontSize(el.fontSize || 10)
               .font(el.bold ? 'Helvetica-Bold' : 'Helvetica')
               .text(val, originX + el.x, originY + el.y, {
                  width: el.maxWidth || undefined,
                  align: el.align || 'left'
               });
          } 
          else if (el.type === 'barcode' && barcodeBuffer) {
            // PDFKit directly consumes the bwip-js memory buffer
            doc.image(barcodeBuffer, originX + el.x, originY + el.y, {
              width: el.width,
              height: el.height
            });
          }
        }
        labelCount++;
      }
    }
    
    // 7. Finalize and send the PDF stream
    doc.end();

  } catch (error) {
    console.error("PDF Generation Error:", error);
    if (!res.headersSent) res.status(500).send('Error generating PDF');
  }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`Hobby Corner API running on port ${PORT}`));
